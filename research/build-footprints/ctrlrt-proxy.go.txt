// Shape-proxy for the build-track's operator and webhook artifacts: a
// controller-runtime manager with a reconciler and a webhook server, which is
// the import graph a kubebuilder scaffold produces.
package main

import (
	corev1 "k8s.io/api/core/v1"
	ctrl "sigs.k8s.io/controller-runtime"
	"sigs.k8s.io/controller-runtime/pkg/client"
	"sigs.k8s.io/controller-runtime/pkg/webhook"
	"sigs.k8s.io/controller-runtime/pkg/webhook/admission"
	"context"
)

type r struct{ client.Client }

func (x *r) Reconcile(ctx context.Context, req ctrl.Request) (ctrl.Result, error) {
	var p corev1.Pod
	return ctrl.Result{}, x.Get(ctx, req.NamespacedName, &p)
}

func main() {
	m, err := ctrl.NewManager(ctrl.GetConfigOrDie(), ctrl.Options{})
	if err != nil {
		panic(err)
	}
	if err := ctrl.NewControllerManagedBy(m).For(&corev1.Pod{}).Complete(&r{m.GetClient()}); err != nil {
		panic(err)
	}
	m.GetWebhookServer().Register("/validate", &webhook.Admission{
		Handler: admission.HandlerFunc(func(context.Context, admission.Request) admission.Response {
			return admission.Allowed("")
		}),
	})
	if err := m.Start(ctrl.SetupSignalHandler()); err != nil {
		panic(err)
	}
}
